param([string]$RunId = 'issue190-logistic-preflight-20260831a')
$ErrorActionPreference = 'Stop'
$PSNativeCommandUseErrorActionPreference = $true
. ./scripts/test-gate-lock.ps1
$tag = "gate-$RunId"
$image = "customer-agent/backend-locks:$tag"
$container = "customer-agent-$RunId-locks"
$ownedImages = @($image,"customer-agent/backend-test:$tag","customer-agent/frontend-test:$tag")
$baseSha = (git rev-parse origin/main).Trim()
$headSha = (git rev-parse HEAD).Trim()
$holder = Enter-TestGateLock -Issue 190 -RunId $RunId -CommandType 'logistic-focused-preflight' -BaseSha $baseSha -HeadSha $headSha -ImageTag $tag
$watch = [Diagnostics.Stopwatch]::StartNew()
$created = $false
$ownedNamespaceConfirmed = $false
$outcome = 'FAIL'
$phase = 'dependency-lock'
try {
    $existing = @(docker image ls --format '{{.Repository}}:{{.Tag}}')
    if (@($ownedImages | Where-Object { $existing -contains $_ }).Count -gt 0) { throw '本RunId镜像已存在，不触碰。' }
    if (docker ps -aq --filter "name=^$container`$") { throw '本RunId容器已存在，不触碰。' }
    $ownedNamespaceConfirmed = $true
    & .local/tools/uv/uv.exe lock --project agent
    & .local/tools/uv/uv.exe sync --project agent --frozen --group calibration
    $phase = 'agent-focused'
    $files = @(git diff --name-only origin/main...HEAD -- '*.py' | Where-Object { $_.StartsWith('agent/') } | ForEach-Object { $_.Substring('agent/'.Length) })
    Push-Location agent
    try {
        & .venv/Scripts/ruff.exe format @files
        & .venv/Scripts/ruff.exe check @files
        & .venv/Scripts/python.exe -m pytest tests/test_knowledge_answerability.py tests/test_knowledge_embedding.py tests/test_knowledge_calibration.py -q
        & .venv/Scripts/pyright.exe --pythonpath .venv/Scripts/python.exe
    } finally { Pop-Location }
    $phase = 'java-canonical-format'
    docker build -f .local/Dockerfile.issue190-locks -t $image backend
    docker create --name $container $image | Out-Null
    $created = $true
    docker cp "${container}:/workspace/gradle.lockfile" backend/gradle.lockfile
    foreach ($file in @(git diff --name-only origin/main...HEAD -- '*.java')) {
        $inside = $file.Substring('backend/'.Length)
        docker cp "${container}:/workspace/$inside" $file
    }
    docker rm -v $container
    $created = $false
    $phase = 'frontend-format'
    & '<USER_HOME>/.cache/codex-runtimes/codex-primary-runtime/dependencies/node/bin/node.exe' .local/tools/prettier/package/bin/prettier.cjs --write frontend/src/workspaces/KnowledgeHybridSearch.tsx frontend/src/workspaces/KnowledgeWorkspace.tsx frontend/e2e/issue190.hybrid-retrieval.spec.ts
    $phase = 'backend-component'
    docker build --target test --tag "customer-agent/backend-test:$tag" backend
    $phase = 'frontend-component'
    docker build --target test --tag "customer-agent/frontend-test:$tag" frontend
    $outcome = 'PASS'
} finally {
    try {
        if ($created) { docker rm -v $container }
        if ($ownedNamespaceConfirmed) {
            $images = @(docker image ls --format '{{.Repository}}:{{.Tag}}')
            foreach ($owned in $ownedImages) { if ($images -contains $owned) { docker image rm $owned } }
        }
        $watch.Stop()
        $directory = ".local/gate-evidence/$RunId"
        New-Item -ItemType Directory -Force $directory | Out-Null
        @{run_id=$RunId; head_sha=$headSha; base_sha=$baseSha; status=$outcome; last_phase=$phase; elapsed_seconds=$watch.Elapsed.TotalSeconds; paid_model_cost_cny=0; working_tree_changed_by_format=$true; model_calibration_or_holdout_run=$false} | ConvertTo-Json | Set-Content "$directory/phase.json" -Encoding utf8
    } finally {
        Exit-TestGateLock $holder
        Write-Output "OWN_LOCK_RELEASED $RunId $outcome"
        Show-TestGateStatus
    }
}
